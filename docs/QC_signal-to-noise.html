# helper function to collapse 9 plasticity combinations into 4 categories we are interested in here
getPlasticityCategory <- function(.cer, .par) {
  if (.cer == 0 & .par == 0) {
    return("ConsZero")
  }
  if (.cer == 0 | .par == 0) {
    return("SingleZero")
  }
  if (.cer  != .par) {
    return("Reversal")
  }
  if (.cer == .par) {
    return("ConsVar")
  }
}
